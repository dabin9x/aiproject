# ==========================================
# 3. 관광지 검색
# ==========================================
st.sidebar.header("🔎 Search Attractions")

search = st.sidebar.text_input(
    "Search in English or Korean",
    placeholder="e.g. Gyeongbokgung"
).strip().lower()

filtered_places = [
    place for place in places
    if search == ""
    or search in place["name"].lower()
    or search in place["korean"].lower()
]

st.metric("Tourist Attractions", len(filtered_places))


# ==========================================
# 4. 지도 생성
# ==========================================
st.subheader("📍 Interactive Seoul Tourist Map")

if filtered_places:
    tourist_map = folium.Map(
        location=[37.5665, 126.9780],
        zoom_start=12,
        tiles="OpenStreetMap"
    )

    # 날짜별 색상
    day_colors = {
        1: "red",
        2: "blue"
    }

    # 날짜별 이동 경로
    for day in [1, 2]:
        day_places = [
            place for place in places
            if place["day"] == day
            and place in filtered_places
        ]

        if len(day_places) > 1:
            folium.PolyLine(
                locations=[
                    [place["lat"], place["lon"]]
                    for place in day_places
                ],
                color=day_colors[day],
                weight=4,
                opacity=0.7,
                tooltip=f"Day {day} Route"
            ).add_to(tourist_map)

    # 관광지 마커
    for place in filtered_places:
        map_url = (
            "https://www.google.com/maps/search/?api=1&query="
            f"{place['lat']},{place['lon']}"
        )

        popup_html = f"""
        <div style="width:230px">
            <h4>{place['name']}</h4>
            <p>{place['korean']}</p>
            <p>{place['description']}</p>
            <p>Day {place['day']}</p>
            <a href="{map_url}" target="_blank">
                Open in Google Maps
            </a>
        </div>
        """

        folium.Marker(
            location=[place["lat"], place["lon"]],
            tooltip=place["name"],
            popup=folium.Popup(popup_html, max_width=300),
            icon=folium.Icon(
                color=day_colors[place["day"]],
                icon="info-sign"
            )
        ).add_to(tourist_map)

    # 지도 범례
    legend_html = """
    <div style="
        position: fixed;
        bottom: 25px;
        left: 25px;
        z-index: 9999;
        background-color: white;
        padding: 10px;
        border: 1px solid grey;
        border-radius: 5px;
        font-size: 14px;">
        <b>Route Legend</b><br>
        <span style="color:red;">●</span> Day 1<br>
        <span style="color:blue;">●</span> Day 2
    </div>
    """

    tourist_map.get_root().html.add_child(
        folium.Element(legend_html)
    )

    st_folium(
        tourist_map,
        use_container_width=True,
        height=550,
        returned_objects=[]
    )

else:
    st.warning("No attractions found. Try another search.")
